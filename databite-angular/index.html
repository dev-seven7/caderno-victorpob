import type { HorarioSemana } from './models';

/**
 * DIFERENCIAL 1 · Status real da loja: "Aberta · até 2h" / "Fechada · abre hoje às 18h".
 *
 * O Cardápio atual mostra só "Fechado". O protótipo diz quando abre, e entende turno que
 * passa da meia-noite (a Databite grava "18:00–23:59" + "00:00–05:00" no mesmo dia,
 * e o "00:00–05:00" é a madrugada seguinte).
 *
 * Funções puras: recebem `agora`, não leem relógio. Testáveis e iguais em Angular ou React.
 */

const MINUTOS_DIA = 1440;
const DIAS_LONGO = ['domingo', 'segunda', 'terça', 'quarta', 'quinta', 'sexta', 'sábado'];

/** Faixa em minutos desde a 00:00 do dia; o fim pode passar de 1440 (madrugada seguinte). */
export type Faixa = readonly [inicio: number, fim: number];

const paraMinutos = (hhmm: string): number => {
  const [h, m] = hhmm.split(':').map(Number);
  return h * 60 + m;
};

/** Turnos de um dia já com a madrugada emendada na noite. */
export function turnosDoDia(semana: HorarioSemana, dia: number): Faixa[] {
  const faixas = semana[((dia % 7) + 7) % 7];
  if (!faixas) return [];
  const brutas = faixas.map(t => [paraMinutos(t.inicio), paraMinutos(t.fim)] as const);
  const viraNoite = brutas.some(([a, b]) => a > 0 && b >= MINUTOS_DIA - 1);

  let madrugada: number | null = null;
  const saida: Array<[number, number]> = [];
  for (const [a, b] of brutas) {
    if (a === 0 && viraNoite) madrugada = b;
    else saida.push([a, b]);
  }
  if (madrugada !== null) {
    for (const t of saida) if (t[1] >= MINUTOS_DIA - 1) t[1] = MINUTOS_DIA + madrugada;
  }
  return saida;
}

/** Minuto em que o turno atual termina (relativo à 00:00 de hoje), ou null se fechada. */
export function fimDoTurnoAtual(semana: HorarioSemana, agora: Date): number | null {
  const m = agora.getHours() * 60 + agora.getMinutes();
  const dia = agora.getDay();
  const hoje = turnosDoDia(semana, dia).find(([a, b]) => m >= a && m <= b);
  if (hoje) return hoje[1];
  const ontem = turnosDoDia(semana, dia - 1).find(([a, b]) => m + MINUTOS_DIA >= a && m + MINUTOS_DIA <= b);
  return ontem ? ontem[1] - MINUTOS_DIA : null;
}

export function estaAberta(semana: HorarioSemana, agora: Date): boolean {
  return fimDoTurnoAtual(semana, agora) !== null;
}

/** Próxima abertura, procurando de 15 em 15 minutos por até 8 dias. */
export function proximaAbertura(semana: HorarioSemana, agora: Date, passoMin = 15, dias = 8): Date | null {
  const t = new Date(agora);
  t.setSeconds(0, 0);
  t.setMinutes(Math.ceil(t.getMinutes() / passoMin) * passoMin);
  const passos = (dias * MINUTOS_DIA) / passoMin;
  for (let i = 0; i < passos; i++) {
    if (estaAberta(semana, t)) return new Date(t);
    t.setMinutes(t.getMinutes() + passoMin);
  }
  return null;
}

/** 1080 → "18h"; 1110 → "18h30"; 1439/1440 → "meia-noite". */
export function formatarHora(minutos: number): string {
  if (minutos === MINUTOS_DIA || minutos === MINUTOS_DIA - 1) return 'meia-noite';
  const m = ((minutos % MINUTOS_DIA) + MINUTOS_DIA) % MINUTOS_DIA;
  const h = Math.floor(m / 60);
  const r = m % 60;
  return r ? `${h}h${String(r).padStart(2, '0')}` : `${h}h`;
}

export interface StatusLoja {
  readonly aberta: boolean;
  readonly texto: string;
}

/** Texto curto do card da loja. */
export function statusLoja(semana: HorarioSemana, agora: Date): StatusLoja {
  const fim = fimDoTurnoAtual(semana, agora);
  if (fim !== null) return { aberta: true, texto: `Aberta · até ${formatarHora(fim)}` };

  const prox = proximaAbertura(semana, agora);
  if (!prox) return { aberta: false, texto: 'Fechada · sem horário' };
  const meiaNoite = (d: Date) => new Date(d.getFullYear(), d.getMonth(), d.getDate()).getTime();
  const dif = Math.round((meiaNoite(prox) - meiaNoite(agora)) / 86_400_000);
  const quando = dif === 0 ? 'hoje' : dif === 1 ? 'amanhã' : DIAS_LONGO[prox.getDay()];
  return { aberta: false, texto: `Fechada · abre ${quando} às ${formatarHora(prox.getHours() * 60 + prox.getMinutes())}` };
}

import { DestroyRef, Directive, ElementRef, OnInit, Renderer2, inject, input } from '@angular/core';

/**
 * DIFERENCIAL 3 · Produto em movimento, sem pesar o celular.
 *
 * O Cardápio atual mostra foto parada (com "Ampliar imagem"). O protótipo mostra um vídeo curto
 * do burger em loop no card e na ficha, com estas travas:
 * - só baixa o vídeo quando o card aparece na tela, e só toca o que está visível;
 * - pausa com a aba escondida;
 * - "menos movimento" (prefers-reduced-motion) ou economia de dados = fica a foto parada;
 * - o navegador escolhe o formato mais leve que abre: AV1 → VP9 (WebM) → H.264 (MP4).
 *   Na bancada de 25/11, AV1 ficou 35–64% menor que VP9; a foto em AVIF, 24% menor que WebP.
 *
 * Uso: <video appVideoProduto base="img/v-mcmanis" poster="img/mcmanis.avif" muted loop playsinline></video>
 */
@Directive({ selector: 'video[appVideoProduto]', standalone: true })
export class VideoProdutoDirective implements OnInit {
  /** Caminho sem extensão; espera <base>.av1.mp4, <base>.webm e <base>.mp4. */
  readonly base = input.required<string>();

  private readonly el = inject<ElementRef<HTMLVideoElement>>(ElementRef);
  private readonly r = inject(Renderer2);
  private readonly destroyRef = inject(DestroyRef);
  private carregado = false;

  ngOnInit(): void {
    const v = this.el.nativeElement;
    v.muted = true; // autoplay só funciona mudo
    v.preload = 'none';
    if (this.semMovimento() || typeof IntersectionObserver === 'undefined') return;

    const obs = new IntersectionObserver(([e]) => {
      if (e.isIntersecting && !document.hidden) {
        this.carregar(v);
        v.play().catch(() => { /* navegador bloqueou: fica o poster */ });
      } else {
        v.pause();
      }
    }, { threshold: 0.5 });
    obs.observe(v);

    const aoEsconder = () => { if (document.hidden) v.pause(); };
    document.addEventListener('visibilitychange', aoEsconder);
    this.destroyRef.onDestroy(() => {
      obs.disconnect();
      document.removeEventListener('visibilitychange', aoEsconder);
    });
  }

  private carregar(v: HTMLVideoElement): void {
    if (this.carregado) return;
    this.carregado = true;
    const fontes: ReadonlyArray<readonly [string, string]> = [
      [`${this.base()}.av1.mp4`, 'video/mp4; codecs="av01.0.05M.08"'],
      [`${this.base()}.webm`, 'video/webm; codecs="vp9"'],
      [`${this.base()}.mp4`, 'video/mp4'],
    ];
    for (const [src, type] of fontes) {
      const s = this.r.createElement('source') as HTMLSourceElement;
      this.r.setAttribute(s, 'src', src);
      this.r.setAttribute(s, 'type', type);
      this.r.appendChild(v, s);
    }
    v.load();
  }

  private semMovimento(): boolean {
    const nav = navigator as Navigator & { connection?: { saveData?: boolean } };
    return matchMedia('(prefers-reduced-motion: reduce)').matches || !!nav.connection?.saveData;
  }
}

import type { Centavos, Produto } from './models';

/**
 * DIFERENCIAL 4 · Busca que perdoa acento + filtros de 1 toque.
 *
 * O Cardápio atual tem busca. O protótipo acrescenta:
 * - busca sem acento e sem maiúscula ("cocó" acha "X-Cocó", "brasileirissimo" acha "Brasileiríssimo");
 * - filtros rápidos: Frango, Costela, Bacon, Até R$ 40, Pra dividir;
 * - número de resultados, que vai na medição (busca com 0 resultado = produto que o cliente quer e não tem).
 */

/** Minúsculas, sem acento: "Cocó" → "coco". */
export function normalizar(texto: string): string {
  return texto.normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase();
}

const contem = (p: Produto, termo: string) => normalizar(`${p.nome} ${p.descricao}`).includes(normalizar(termo));

export interface FiltroRapido {
  readonly id: string;
  readonly rotulo: string;
  readonly aceita: (p: Produto) => boolean;
}

/**
 * Filtros por ingrediente leem a descrição hoje. No produto final, o ideal é a Databite mandar
 * tags por produto (frango, costela, vegetariano…) para não depender do texto (pergunta B3).
 */
export function filtrosPadrao(tetoBarato: Centavos = 4000): FiltroRapido[] {
  return [
    { id: 'frango', rotulo: 'Frango', aceita: p => contem(p, 'frango') },
    { id: 'costela', rotulo: 'Costela', aceita: p => contem(p, 'costela') },
    { id: 'bacon', rotulo: 'Bacon', aceita: p => contem(p, 'bacon') },
    { id: 'barato', rotulo: `Até R$ ${tetoBarato / 100}`, aceita: p => p.preco <= tetoBarato },
    { id: 'dividir', rotulo: 'Pra dividir', aceita: p => p.categoriaId === 'dividir' },
  ];
}

/** Busca + filtros ligados (todos precisam valer). */
export function filtrarCardapio(
  produtos: ReadonlyArray<Produto>,
  termo: string,
  filtrosAtivos: ReadonlyArray<FiltroRapido>,
): Produto[] {
  const t = termo.trim();
  return produtos.filter(p => (!t || contem(p, t)) && filtrosAtivos.every(f => f.aceita(p)));
}

import type { Centavos, GrupoOpcoes, Produto, Selecao } from './models';

/**
 * DIFERENCIAL 5 · Ficha do produto que guia a escolha.
 *
 * Ficamos com o melhor de cada lado:
 * - Do Cardápio atual, o MODELO: grupos com mínimo, máximo e obrigatório, vindos da API.
 *   O protótipo tinha regras fixas por tipo de produto e isso não escala.
 * - Do protótipo, a EXPERIÊNCIA: o botão diz o que falta ("Escolhe a bebida",
 *   "Escolhe mais 2"), o preço atualiza na hora e um grupo condicional só aparece quando faz sentido.
 */

const total = (sel: Selecao, grupoId: string): number =>
  Object.values(sel[grupoId] ?? {}).reduce((a, b) => a + b, 0);

/**
 * Seleção inicial da ficha (protótipo, 25/11 · "o que o atual faz melhor, faça melhor também"):
 * - grupo obrigatório de UMA opção só ("Pão", "Pegue seu Ogrostela") já vem marcado e aparece como "Já vem com";
 * - grupo obrigatório de 1 escolha em que só UMA opção é grátis e as outras custam (Fome normal × Fome de Ogro,
 *   Batata P × G) já vem com a grátis: o cliente só toca se quiser mudar. Nunca pré-marca opção paga.
 * Bebida (todas grátis) continua sem pré-seleção: o sabor é escolha do cliente.
 */
export function selecaoPadrao(produto: Produto): Selecao {
  const sel: Record<string, Record<string, number>> = {};
  for (const g of produto.grupos) {
    if (g.visivelSe || g.min < 1) continue;
    if (g.opcoes.length === 1 && g.max === 1) { sel[g.id] = { [g.opcoes[0].id]: 1 }; continue; }
    if (g.min === 1 && g.max === 1) {
      const gratis = g.opcoes.filter(o => o.preco === 0);
      if (gratis.length === 1 && g.opcoes.length > 1) sel[g.id] = { [gratis[0].id]: 1 };
    }
  }
  return sel;
}

/** Grupos que vêm resolvidos (uma opção só): a tela mostra numa linha "Já vem com", sem pedir toque. */
export const jaVemCom = (produto: Produto): ReadonlyArray<GrupoOpcoes> =>
  produto.grupos.filter(g => g.min >= 1 && g.max === 1 && g.opcoes.length === 1 && !g.visivelSe);

/** Grupos que aparecem agora (respeita `visivelSe`, ex.: bebida só depois de marcar um combo). */
export function gruposVisiveis(produto: Produto, sel: Selecao): GrupoOpcoes[] {
  return produto.grupos.filter(g => !g.visivelSe || (sel[g.visivelSe.grupoId]?.[g.visivelSe.opcaoId] ?? 0) > 0);
}

/** Texto do que falta, na ordem dos grupos. Vazio = pode adicionar. */
export function oQueFalta(produto: Produto, sel: Selecao): string {
  for (const g of gruposVisiveis(produto, sel)) {
    const n = total(sel, g.id);
    if (n < g.min) {
      if (g.min === 1 && g.max === 1) return `Escolhe: ${g.titulo}`;
      return `Escolhe mais ${g.min - n}`;
    }
  }
  return '';
}

/** Pode somar mais uma unidade desta opção sem passar do máximo do grupo? */
export function podeAumentar(grupo: GrupoOpcoes, sel: Selecao, opcaoId: string): boolean {
  if (total(sel, grupo.id) >= grupo.max) return false;
  return grupo.repetir || (sel[grupo.id]?.[opcaoId] ?? 0) === 0;
}

/**
 * Aplica um toque numa opção e devolve a NOVA seleção (imutável, bom para signals/OnPush).
 * - Rádio (max 1, sem repetir): marcar outra opção substitui a anterior; tocar de novo desmarca
 *   quando o grupo é opcional (min 0), como um checkbox.
 * - Desmarcou a opção que abria um grupo condicional? O grupo sai junto: ninguém paga por grupo escondido.
 */
export function alternar(produto: Produto, grupo: GrupoOpcoes, sel: Selecao, opcaoId: string, delta: 1 | -1): Selecao {
  const atual: Record<string, number> = { ...(sel[grupo.id] ?? {}) };
  if (grupo.max === 1 && !grupo.repetir) {
    const marcada = (atual[opcaoId] ?? 0) > 0;
    const proximo = delta === 1 && !(marcada && grupo.min === 0) ? { [opcaoId]: 1 } : {};
    return normalizar(produto, { ...sel, [grupo.id]: proximo });
  }
  if (delta === 1 && !podeAumentar(grupo, sel, opcaoId)) return sel;
  const novo = Math.max(0, (atual[opcaoId] ?? 0) + delta);
  if (novo === 0) delete atual[opcaoId];
  else atual[opcaoId] = novo;
  return normalizar(produto, { ...sel, [grupo.id]: atual });
}

/** Remove escolhas de grupos que deixaram de ser visíveis. */
export function normalizar(produto: Produto, sel: Selecao): Selecao {
  const visiveis = new Set(gruposVisiveis(produto, sel).map(g => g.id));
  return Object.fromEntries(Object.entries(sel).filter(([id]) => visiveis.has(id)));
}

/** Preço unitário estimado: base + soma das opções × quantidade. O servidor confirma (B6). */
export function precoUnitario(produto: Produto, sel: Selecao): Centavos {
  let valor = produto.preco;
  for (const g of gruposVisiveis(produto, sel)) {
    for (const o of g.opcoes) valor += o.preco * (sel[g.id]?.[o.id] ?? 0);
  }
  return valor;
}

/** Chave estável do item na sacola: mesma escolha = mesmo item (soma quantidade). */
export function chaveItem(produtoId: string, sel: Selecao, ajustes: ReadonlyArray<string> = []): string {
  const ordenada = Object.keys(sel).sort().map(g => [g, Object.entries(sel[g]).sort(([a], [b]) => a.localeCompare(b))]);
  return `${produtoId}|${JSON.stringify(ordenada)}|${[...ajustes].sort().join(',')}`;
}

      const existente = semAntigo.findIndex(x => x.chave === novo.chave);
      if (existente >= 0) {
        return semAntigo.map((x, k) => (k === existente ? { ...x, quantidade: x.quantidade + novo.quantidade } : x));
      }
      const pos = lista.findIndex(x => x.chave === chaveAntiga);
      const copia = [...semAntigo];
      copia.splice(pos < 0 ? copia.length : pos, 0, novo);
      return copia;
    });
  }

  alterarQuantidade(chave: string, delta: number): void {
    const lista = this._itens();
    const indice = lista.findIndex(x => x.chave === chave);
    if (indice < 0) return;

      copia.splice(Math.min(r.indice, copia.length), 0, r.item);
      return copia;
    });
    return true;
  }

    this.removido = null;
    this._itens.set([]);
  }

    lojaId: m?.[1] ?? null,
    produtoRef: m ? slug(m[2] ?? m[3]) : null,

 * - Meta só com consentimento de Marketing, com o MESMO event_id (deduplicação com a CAPI).

  private readonly janela = inject(DOCUMENT).defaultView as JanelaComTags | null;
  private readonly sessao = this.gerarSessao();
  private seq = 0;
  private utm: Record<string, string> = {};

  readonly consentimento = signal<Consentimento>({ analise: false, marketing: false });
  /** Loja atual, vira store_id em todo evento ("ogro_botafogo"). */
  readonly lojaId = signal<string | null>(null);

  definirOrigemCampanha(utm: Readonly<Record<string, string>>): void {
    this.utm = { ...utm };
  }

  track(evento: string, params: Record<string, unknown> = {}, ecommerce?: { value?: number; items?: ItemGA4[] }): string {
    const eventId = `${this.sessao}_${++this.seq}`;
    const { analise, marketing } = this.consentimento();
    if (!analise || !this.janela) return eventId;

    const loja = this.lojaId();
    const payload: Record<string, unknown> = {
      event: evento,
      event_id: eventId,
      session_id: this.sessao,
      store_id: loja ? `ogro_${loja.replace(/-/g, '_')}` : undefined,
      ...this.utm,
      ...params,
    };
    if (ecommerce) payload['ecommerce'] = { currency: 'BRL', ...ecommerce };

    const dl = (this.janela.dataLayer ??= []);
    if (ecommerce) dl.push({ ecommerce: null }); // limpa o objeto anterior (recomendação do GA4)
    dl.push(this.semVazios(payload));

    const meta = META.get(evento);
    if (marketing && meta && typeof this.janela.fbq === 'function') {
      const items = ecommerce?.items ?? [];
      this.janela.fbq('track', meta, {
        content_type: 'product',
        content_ids: items.map(i => i.item_id),
        contents: items.map(i => ({ id: i.item_id, quantity: i.quantity ?? 1 })),
        value: ecommerce?.value,
        currency: 'BRL',
      }, { eventID: eventId });
    }
    return eventId;
  }

  private semVazios(o: Record<string, unknown>): Record<string, unknown> {
    return Object.fromEntries(Object.entries(o).filter(([, v]) => v !== undefined));
  }

  private gerarSessao(): string {
    const c = this.janela?.crypto;
    return c?.randomUUID ? c.randomUUID().slice(0, 13) : Date.now().toString(36);
  }

@Injectable({ providedIn: 'root' })
export class ImpressoesService {
  private readonly analytics = inject(AnalyticsService);
  private readonly vistos = new Set<string>();
  private readonly fila = new Map<string, ItemGA4[]>();
  private timer: ReturnType<typeof setTimeout> | undefined;

  registrar(lista: string, item: ItemGA4): void {
    const chave = `${lista}|${item.item_id}`;
    if (this.vistos.has(chave)) return;
    this.vistos.add(chave);
    this.fila.set(lista, [...(this.fila.get(lista) ?? []), item]);
    clearTimeout(this.timer);
    this.timer = setTimeout(() => this.enviar(), 1200);
  }

  /** Chamar ao sair da tela (troca de rota) para fechar a visita. */
  enviar(): void {
    for (const [lista, itens] of this.fila) {
      for (let i = 0; i < itens.length; i += 20) {
        this.analytics.track('view_item_list', { item_list_id: lista }, { items: itens.slice(i, i + 20) });
      }
    }
    this.fila.clear();
  }

  novaVisita(): void {
    this.enviar();
    this.vistos.clear();
  }

/** Nomes escolhidos num grupo (pelo título), na ordem das opções do cardápio. */
function escolhidos(produto: Produto, sel: Selecao, titulo: string): string[] {
  const g = produto.grupos.find(x => semAcento(x.titulo) === semAcento(titulo));
  if (!g) return [];
  return g.opcoes.filter(o => (sel[g.id]?.[o.id] ?? 0) > 0).map(o => o.nome);
}

export function montarBurger(produto: Produto, sel: Selecao): Montagem {
  const carne = escolhidos(produto, sel, 'Carne')[0];
  const queijos = escolhidos(produto, sel, 'Queijos'), molhos = escolhidos(produto, sel, 'Molhos'), compl = escolhidos(produto, sel, 'Complementos');
  const arqs = (xs: ReadonlyArray<string>) => xs.map(arquivoDe).filter((a): a is ArquivoCamada => !!a);

  const ordem: Array<ArquivoCamada | 'vazio'> = ['pao-topo', ...arqs(molhos.slice(1)), ...salada, ...arqs(queijos),

  const L = 780; let y = 0; const camadas: CamadaPosicionada[] = [];
  ordem.forEach((a, k) => {

    else if (a === 'vazio') y += 8;

    y += h;
  });
  const itens = [carne, ...queijos, ...molhos, ...compl].filter(Boolean);
  return { camadas, largura: L, altura: y, temCarne: !!carne, descricao: itens.length ? `Teu burger: ${itens.join(', ')}` : 'Monte teu burger' };
}

  readonly base = input.required<string>();
  readonly nome = input.required<string>();
  readonly fechar = output<void>();

  protected readonly perto = signal(false);
  private readonly caixa = viewChild.required<ElementRef<HTMLElement>>('caixa');
  private readonly fecharBt = viewChild.required<ElementRef<HTMLButtonElement>>('fecharBt');

  constructor() {
    afterNextRender(() => this.fecharBt().nativeElement.focus({ preventScroll: true }));
  }

  protected alternar(ev: MouseEvent): void {
    ev.stopPropagation();
    const img = ev.target as HTMLImageElement, r = img.getBoundingClientRect();
    const fx = (ev.clientX - r.left) / r.width, fy = (ev.clientY - r.top) / r.height;
    this.perto.update(v => !v);
    if (this.perto()) {
      requestAnimationFrame(() => {
        const c = this.caixa().nativeElement;
        c.scrollLeft = fx * c.scrollWidth - c.clientWidth / 2;
        c.scrollTop = fy * c.scrollHeight - c.clientHeight / 2;
      });
    }
  }

  protected fora(ev: MouseEvent): void {
    const alvo = ev.target as HTMLElement;
    if (alvo.classList.contains('zoom') || alvo.classList.contains('zoom__foto')) this.fechar.emit();
  }
}
